cmake_minimum_required(VERSION 3.20)
project(atto_shared_tenant_bytes LANGUAGES CXX)
include(CTest)
option(ATTO_COMMON_SANITIZERS "Run native allocator tests under ASan/UBSan" ON)
add_library(atto_tenant_bytes_v1 INTERFACE)
target_include_directories(atto_tenant_bytes_v1 INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(atto_tenant_bytes_v1 INTERFACE cxx_std_20)
add_executable(atto_tenant_bytes_v1_test tests/tenant_bytes_v1.cpp)
target_link_libraries(atto_tenant_bytes_v1_test PRIVATE atto_tenant_bytes_v1)
find_package(Threads REQUIRED)
target_link_libraries(atto_tenant_bytes_v1_test PRIVATE Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(atto_tenant_bytes_v1_test PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
  if(ATTO_COMMON_SANITIZERS)
    target_compile_options(atto_tenant_bytes_v1_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(atto_tenant_bytes_v1_test PRIVATE -fsanitize=address,undefined)
  endif()
endif()
add_test(NAME native_tenant_bytes_v1 COMMAND atto_tenant_bytes_v1_test)
